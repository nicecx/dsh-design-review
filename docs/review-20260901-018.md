```js
// core.js（纯函数，可单测）
export function isSkillDoc(filePath) {
  const base = path.basename(filePath).toLowerCase()
  if (base === 'skill.md') return true
  return /[\/\\]skills[\/\\]/.test(filePath) && base.endsWith('.skill.md')
}
export function gateByType(type, content) {
  // 统一关卡：design=复用+模板；skill=仅复用；lesson=跳过
  if (type === 'skill') return checkReuseSection(content)      // 只查复用
  if (type === 'design') {
    const r = checkReuseSection(content)
    if (!r.ok) return r
    const t = checkTemplateSections(content)
    return t.ok ? r : { ok: false, reason: `设计模板缺章 ${t.missing.join('、')}`, references: r.references }
  }
  return { ok: true, reason: 'lesson 豁免', references: { index: [], awesome: [], github: [] } }
}
```
- `submit()` 与 auto-submit 均调用 `gateByType`（不再内联双检查）
- I1-I4 直接测 `gateByType` 纯函数（I1 skill 含复用节→ok；I2 skill 无节→拒；I3 逃逸→ok；I4 design 缺章→拒）
